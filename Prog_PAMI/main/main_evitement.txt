// ══════════════════════════════════════════════════════════
//  MULTIPLEXEUR I2C
// ══════════════════════════════════════════════════════════
bool tcaselect(uint16_t i)
{
  if (i > 7)
    return false;
  Wire.beginTransmission(TCAADDR);
  Wire.write(1 << i);
  if (Wire.endTransmission() != 0)
    return false;
  delay(2);
  return true;
}

bool tcaselectAll()
{
  Wire.beginTransmission(TCAADDR);
  Wire.write(0x0F);
  if (Wire.endTransmission() != 0)
    return false;
  delay(2);
  return true;
}

// ══════════════════════════════════════════════════════════
//  RÉCEPTION TCP
// ══════════════════════════════════════════════════════════
void gestion_tcp()
{
  WiFiClient client = server.accept();
  if (!client)
    return;

  Serial.println("[TCP] Client connecté.");

  String message = "";
  unsigned long timeout = millis();
  while (client.connected() && millis() - timeout < 5000)
  {
    while (client.available())
    {
      char c = client.read();
      if (c == '\n')
        goto done;
      message += c;
      timeout = millis();
    }
  }
done:
  message.trim();
  client.stop();

  Serial.print("[TCP] Reçu : ");
  Serial.println(message);

  JsonDocument doc;
  DeserializationError error = deserializeJson(doc, message);
  if (error)
  {
    Serial.print("[JSON] Erreur : ");
    Serial.println(error.c_str());
    return;
  }

  couleur = doc["couleur"] | 0;
  PAMI_debut_match = doc["PAMI_debut_match"] | false;

  Serial.print("[MATCH] Couleur : ");
  Serial.println(couleur);
  Serial.print("[MATCH] PAMI_debut_match : ");
  Serial.println(PAMI_debut_match);

  if (PAMI_debut_match && !timer_85_demarre)
  {
    Serial.println("[MATCH] Signal reçu ! Timer 85s démarré...");
    temps_debut_match = millis();
    timer_85_demarre = true;
  }

  if (!PAMI_debut_match && robot_demarre_par_wifi)
  {
    Serial.println("[MATCH] Arrêt reçu !");
    robot_demarre_par_wifi = false;
    timer_85_demarre = false;
    stepperL->stopMove();
    stepperR->stopMove();
    etat = ATTENTE;
    mode_sequence = false;
  }
}

int lire_capteur(int id)
{
  if (id == 0)
    return ok0 ? (int)sensor0.readRangeContinuousMillimeters() : 0;
  if (id == 1)
    return ok1 ? (int)sensor1.readRangeContinuousMillimeters() : 0;
  if (id == 2)
    return ok2 ? (int)sensor2.readRangeContinuousMillimeters() : 0;
  if (id == 3)
    return ok3 ? (int)sensor3.readRangeContinuousMillimeters() : 0;

  return 8888;
}

void reset_moteurs()
{
  stepperL->setCurrentPosition(0);
  stepperR->setCurrentPosition(0);
}

void maj_pos()
{
  float mmL = stepperL->getCurrentPosition() / k_LIN;
  float mmR = stepperR->getCurrentPosition() / (k_LIN * CORRECTION_DERIVE);
  float mm = (mmL + mmR) / 2.0;
  robot_x += mm * cos(robot_a * PI / 180.0);
  robot_y += mm * sin(robot_a * PI / 180.0);
}

// Sauvegarde la position puis remet les encodeurs à zéro
// À appeler AVANT tout reset_moteurs() pour ne jamais perdre de déplacement
void maj_et_reset()
{
  maj_pos();
  reset_moteurs();
}

// Stoppe les moteurs, attend l'arrêt complet, puis met à jour la position
void stop_moteurs_et_maj()
{
  stepperL->stopMove();
  stepperR->stopMove();
  while (stepperL->isRunning() || stepperR->isRunning())
    delay(1);
  maj_et_reset();
}

// Vérifie si un obstacle est détecté sur les capteurs avant/gauche/droite
bool obstacle_detecte()
{
  int dG = lire_capteur(1);
  int dD = lire_capteur(2);
  return ((dG != 8888 && dG < DISTANCE_ARRET) || (dD != 8888 && dD < DISTANCE_ARRET));
}

// ══════════════════════════════════════════════════════════
//  LOGIQUE DE MOUVEMENT
// ══════════════════════════════════════════════════════════
void gestion_etats()
{
  // FastAccelStepper est piloté par timer hardware : plus besoin de stepperL.run() ici

  switch (etat)
  {
  case ATTENTE:
    break;

  case CALCUL_ROT:
  {
    float dx = cible_x - robot_x;
    float dy = cible_y - robot_y;
    float dist = sqrt(dx * dx + dy * dy);

    if (dist < 20)
    {
      etat = AVANCE;
      return;
    }

    float angle_vise = atan2(dx, dy) * 180.0 / PI;
    float rot = angle_vise - robot_a;
    while (rot > 180)
      rot -= 360;
    while (rot <= -180)
      rot += 360;

    reset_moteurs();
    long pas = (long)(rot * k_ROT);

    // Avec FastAccelStepper : setSpeedInHz() avant move()
    stepperL->setSpeedInHz((uint32_t)VITESSE_MAX);
    stepperR->setSpeedInHz((uint32_t)VITESSE_MAX);

    stepperL->move(-pas);
    stepperR->move(pas);

    robot_a += rot;
    while (robot_a > 180)
      robot_a -= 360;
    while (robot_a <= -180)
      robot_a += 360;

    etat = ROTATION;
  }
  break;

  case ROTATION:
    // distanceToGo() == 0 remplacé par !isRunning()
    if (!stepperL->isRunning() && !stepperR->isRunning())
    {
      timer_pause = millis();
      etat = PAUSE_ROT;
    }
    break;

  case PAUSE_ROT:
    if (millis() - timer_pause > 100)
      etat = CALCUL_AV;
    break;

  case CALCUL_AV:
  {
    float dx = cible_x - robot_x;
    float dy = cible_y - robot_y;
    float dist = sqrt(dx * dx + dy * dy);

    reset_moteurs();

    long pas_G = (long)(dist * k_LIN);
    long pas_D = (long)(dist * k_LIN * CORRECTION_DERIVE);

    stepperL->setSpeedInHz((uint32_t)VITESSE_MAX);
    stepperR->setSpeedInHz((uint32_t)(VITESSE_MAX * CORRECTION_DERIVE));

    stepperL->move(pas_G);
    stepperR->move(pas_D);

    timer_debut_mouvement = millis();
    timer_capteur = millis();
    compteur_obs = 0;
    etat = AVANCE;
  }
  break;

  case AVANCE:
    if (!stepperL->isRunning() && !stepperR->isRunning())
    {
      robot_x = cible_x;
      robot_y = cible_y;

      if (mode_sequence)
      {
        Serial.print("Arrive zone ");
        Serial.println(zone_actuelle);
        timer_pause = millis();
        etat = PAUSE_SEQUENCE;
      }
      else
      {
        Serial.println("Fini.");
        etat = ATTENTE;
      }
    }
    else if (millis() - timer_capteur > 50)
    {
      timer_capteur = millis();
      if (millis() - timer_debut_mouvement > 500)
      {
        int dG = lire_capteur(1);
        int dD = lire_capteur(2);
        if ((dG != 8888 && dG < DISTANCE_ARRET) || (dD != 8888 && dD < DISTANCE_ARRET))
        {
          compteur_obs++;
          if (compteur_obs >= 2)
          {
            etat = OBS_STOP;
            Serial.println("OBSTACLE !");
          }
        }
        else
        {
          compteur_obs = 0;
        }
      }
    }
    break;

  // --- LOGIQUE DE SEQUENCE ---
  case PAUSE_SEQUENCE:
    if (millis() - timer_pause > 1000)
    {
      zone_actuelle++;
      if (zone_actuelle <= ZONE_FIN)
      {
        aller_a_la_zone(zone_actuelle);
      }
      else
      {
        Serial.println("SEQUENCE TERMINEE (1 a 5).");
        mode_sequence = false;
        etat = ATTENTE;
      }
    }
    break;

  // OBSTACLE
  case OBS_STOP:
    // stopMove() décélère proprement ; on attend l'arrêt complet
    stepperL->stopMove();
    stepperR->stopMove();
    while (stepperL->isRunning() || stepperR->isRunning())
      delay(1);
    maj_pos();
    reset_moteurs();
    etat = OBS_ANALYSE;
    break;

  case OBS_ANALYSE:
  {
    int dG = lire_capteur(1);
    int dD = lire_capteur(2);
    if (couleur == 1)
    {
      sens_esquive = 1;
    }
    else
    {
      sens_esquive = -1;
    }
    etat = OBS_TOURNE;
  }
  break;

  case OBS_TOURNE:
    if (!stepperL->isRunning() && !stepperR->isRunning())
    {
      int dG = lire_capteur(1);
      int dD = lire_capteur(2);
      if (dG > DISTANCE_LOIN && dD > DISTANCE_LOIN)
      {
        reset_moteurs();
        float dist_esquive = (DISTANCE_ARRET + 50.0) * 1.5;
        long pas = (long)(dist_esquive * k_LIN);
        stepperL->setSpeedInHz((uint32_t)VITESSE_MAX);
        stepperR->setSpeedInHz((uint32_t)VITESSE_MAX);
        stepperL->move(pas);
        stepperR->move(pas);
        timer_debut_mouvement = millis();
        etat = OBS_AVANCE_SORTIE;
      }
      else
      {
        reset_moteurs();
        long p = (long)(PAS_ROTATION * k_ROT * sens_esquive);
        stepperL->setSpeedInHz((uint32_t)VITESSE_MAX);
        stepperR->setSpeedInHz((uint32_t)VITESSE_MAX);
        stepperL->move(-p);
        stepperR->move(p);
        angle_scan_total += PAS_ROTATION;
        robot_a += PAS_ROTATION * sens_esquive;
      }
    }
    break;

  case OBS_AVANCE_SORTIE:
    if (!stepperL->isRunning() && !stepperR->isRunning())
    {
      maj_pos();
      etat = CALCUL_ROT;
    }
    break;
  }
}

void gestion_entrees()
{
  if (Serial.available() > 0)
  {
    char c = Serial.read();
    if (c == 'S' || c == 's')
    {
      stepperL->stopMove();
      stepperR->stopMove();
      reset_moteurs();
      etat = ATTENTE;
      mode_sequence = false;
      robot_demarre_par_wifi = false;
      timer_85_demarre = false;
      Serial.println("STOP URGENCE");
    }
  }
}

// ══════════════════════════════════════════════════════════

void setup()
{
  Serial.begin(115200);
  delay(10);
  Serial.println("\nInitialisation...\n");

  // LCD init anticipée pour afficher les messages d'init
  Wire.begin(I2C_SDA, I2C_SCL);
  Wire.setClock(100000);
  lcd.begin(16, 2);
  lcd.clear();
  lcd.setRGB(255, 255, 255);
  lcd.setCursor(0, 0);
  lcd.print("Init en cours...");
  lcd.setCursor(0, 1);
  lcd.print("                ");

  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(vBat, INPUT_PULLUP);
  pinMode(stby, OUTPUT);
  digitalWrite(stby, LOW);
  digitalWrite(enablePin, LOW);
  pinMode(enablePin, OUTPUT_OPEN_DRAIN);
  pinMode(mode1Pin, OUTPUT);
  pinMode(mode2Pin, OUTPUT);
  pinMode(dirPin_L, OUTPUT);
  pinMode(dirPin_R, OUTPUT);
  pinMode(stepPin_L, OUTPUT);
  pinMode(stepPin_R, OUTPUT);
  digitalWrite(dirPin_L, HIGH);
  digitalWrite(dirPin_R, HIGH);
  digitalWrite(stepPin_L, HIGH);
  digitalWrite(stepPin_R, HIGH);
  digitalWrite(mode1Pin, HIGH);
  digitalWrite(mode2Pin, HIGH);

  delay(500);
  digitalWrite(stby, HIGH);
  delay(100);
  digitalWrite(enablePin, HIGH);

  engine.init();
  stepperL = engine.stepperConnectToPin(stepPin_L);
  if (stepperL)
  {
    stepperL->setDirectionPin(dirPin_L, true);
    stepperL->setAcceleration((uint32_t)ACCELERATION);
    stepperL->setSpeedInHz((uint32_t)VITESSE_MAX);
  }
  stepperR = engine.stepperConnectToPin(stepPin_R);
  if (stepperR)
  {
    stepperR->setDirectionPin(dirPin_R, false);
    stepperR->setAcceleration((uint32_t)ACCELERATION);
    stepperR->setSpeedInHz((uint32_t)VITESSE_MAX);
  }

  // Wire déjà initialisé au début du setup

  tcaselect(0);

  if (sensor0.init())
  {
    Serial.printf("Sensor0 détecté en 0x29\n");
    sensor0.setAddress(0x40);
    ok0 = true;

    Serial.printf("Sensor0 non détecté en 0x29\n");
    if (sensor0.init())
    {
      Serial.printf("Sensor0 détecté en 0x40\n");
      ok0 = true;
    }
    else
    {
      Serial.printf("ERREUR : Sensor0 non détecté\n");
    }

  tcaselect(1);

  if (sensor1.init())
  {
    Serial.printf("Sensor1 détecté en 0x29\n");
    sensor1.setAddress(0x41);
    ok1 = true;
  }
  else
  {
    sensor1.setAddress(0x41, false);
    Serial.printf("Sensor1 non détecté en 0x29\n");
    if (sensor1.init())
    {
      Serial.printf("Sensor1 détecté en 0x41\n");
      ok1 = true;
    }
    else
    {
      Serial.printf("ERREUR : Sensor1 non détecté\n");
    }
  }
  sensor1.setTimeout(500);
  sensor1.startContinuous();

  tcaselect(2);

  if (sensor2.init())
  {
    Serial.printf("Sensor2 détecté en 0x29\n");
    sensor2.setAddress(0x42);
    ok2 = true;
  }
  else
  {
    sensor2.setAddress(0x42, false);
    Serial.printf("Sensor2 non détecté en 0x29\n");
    if (sensor2.init())
    {
      Serial.printf("Sensor2 détecté en 0x42\n");
      ok2 = true;
    }
    else
    {
      Serial.printf("ERREUR : Sensor2 non détecté\n");
    }

  sensor2.setTimeout(500);
  sensor2.startContinuous();

  tcaselect(3);

    Serial.printf("Sensor3 détecté en 0x29\n");
    sensor3.setAddress(0x43);
    ok3 = true;
  }
  else
  {
    sensor3.setAddress(0x43, false);

  Serial.printf("[WiFi] Connexion à %s ...\n", SSID);

  WiFi.mode(WIFI_STA);
  WiFi.setTxPower(WIFI_POWER_19_5dBm);
  WiFi.begin(SSID, PASSWORD);

  // Étape 1 : Attendre l'association WiFi
  int tentatives = 0;
  const int MAX_TENTATIVES = 40;

  while (WiFi.status() != WL_CONNECTED && tentatives < MAX_TENTATIVES)
  {
    delay(500);
    Serial.print(".");
    tentatives++;
  }

  if (WiFi.status() != WL_CONNECTED)
  {
    Serial.println("\n[WiFi] ERREUR : Association WiFi impossible !");
    lcd.setCursor(0, 0);
    lcd.setRGB(255, 0, 0);
    lcd.print("WiFi ERREUR !   ");
  }
  else
  {
    Serial.println("\n[WiFi] Association WiFi réussie !");

    // Étape 2 : Attendre l'attribution IP (DHCP)
    Serial.println("[WiFi] Attente de l'adresse IP...");
    int timeout_ip = 0;
    while (WiFi.localIP() == IPAddress(0, 0, 0, 0) && timeout_ip < 20)
    {
      delay(500);
      Serial.print(".");
      timeout_ip++;
    }

    // Vérification finale
    IPAddress ip = WiFi.localIP();
    if (ip == IPAddress(0, 0, 0, 0))
    {
      Serial.println("\n[WiFi] ERREUR : Pas d'adresse IP obtenue !");
      Serial.println("[WiFi] Le DHCP n'a pas répondu.");
      Serial.println("[WiFi] Solutions possibles :");
      Serial.println("  - Redémarrez le routeur");
      Serial.println("  - Vérifiez que le DHCP est activé");
      Serial.println("  - Essayez une IP statique");

      lcd.setCursor(0, 0);
      lcd.setRGB(255, 165, 0);
      lcd.print("WiFi: Pas d'IP! ");
    }
    else
    {
      Serial.printf("\n[WiFi] ✓ CONNEXION RÉUSSIE !\n");
      Serial.printf("[WiFi] IP    : %s\n", ip.toString().c_str());
      Serial.printf("[WiFi] Gateway: %s\n", WiFi.gatewayIP().toString().c_str());
      Serial.printf("[WiFi] Subnet : %s\n", WiFi.subnetMask().toString().c_str());
      Serial.printf("[WiFi] DNS   : %s\n", WiFi.dnsIP().toString().c_str());
      Serial.printf("[WiFi] Canal : %d\n", WiFi.channel());
      Serial.printf("[WiFi] RSSI  : %d dBm\n", WiFi.RSSI());
      Serial.printf("[WiFi] MAC   : %s\n", WiFi.macAddress().c_str());

      // LCD : WiFi connecté, en attente du signal JSON (blanc)
      lcd.setRGB(255, 255, 255);
      lcd.setCursor(0, 0);
      lcd.print("WiFi connecte   ");
      lcd.setCursor(0, 1);
      lcd.print("En attente signal");
    }
  }

  server.begin();
  Serial.println("[TCP] Serveur démarré sur port 5002");
}

void loop()
{
  // ── 1. Écoute TCP ─────────────────────────────────────
  gestion_tcp();

  // ── 2. Timer 85 secondes ──────────────────────────────
  if (timer_85_demarre && !robot_demarre_par_wifi)
  {
    unsigned long elapsed = millis() - temps_debut_match;
    if (elapsed >= DELAI_DEPART)
    {
      Serial.println("[MATCH] 85 secondes écoulées → DÉPART !");
      demarrer_sequence_couleur();
    }
    else
    {
      static unsigned long last_print = 0;
      if (millis() - last_print > 1000)
      {
        last_print = millis();
        Serial.printf("[MATCH] Départ dans %lu secondes...\n",
                      (DELAI_DEPART - elapsed) / 1000);
      }
    }
  }

  // ── 3. Logique robot ──────────────────────────────────
  if (robot_demarre_par_wifi)
  {
    gestion_entrees();
    gestion_etats();
    afficher_lcd_match();
  }
  else
  {
    afficher_lcd_attente();
  }

  delay(10);
}
